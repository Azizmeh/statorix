# build et démarre superset
docker compose up --build superset -d

# démarrer tous les contenaires
docker compose up -d

# arrêter tous les contenaires 
docker compose down

# arrêter un contenaire docker, exemple superset
docker stop superset

# démarrer nginx
docker compose up nginx -d

# se connecter à superset dans docker 
docker exec -it superset bash

# build de tous les contenaires dans docker
docker compose build --no-cache

# afficher la log superset
docker logs -f superset
